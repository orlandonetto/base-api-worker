import { celebrate, Joi, Segments } from 'celebrate'
import { NextFunction, Request, Response } from 'express'

import AppError from '../../errors/AppError'
import { isEmpty } from '../../helpers/array-helper'
import { isNullOrUndefined } from '../../helpers/object-helper'
import {
  objectId,
  paginationBaseSchema,
} from '../../helpers/validations-helper'
import { HttpStatus } from '../../types/global.enums'
import { findExampleByID, findExamplesByIDs } from './examples.dao'

const paramsBaseSchema = {
  exampleID: objectId(Joi).required(),
}

const validateCreateExampleSchema = celebrate({
  [Segments.BODY]: {
    name: Joi.string().required(),
  },
})

const validateFetchExamplesSchema = celebrate({
  [Segments.QUERY]: {
    ...paginationBaseSchema,
    name: Joi.string().optional(),
    exampleIDs: Joi.array().items(objectId(Joi)).min(1).optional(),
  },
})

const validateFetchExampleSchema = celebrate({
  [Segments.PARAMS]: paramsBaseSchema,
})

const validateEditExampleSchema = celebrate({
  [Segments.PARAMS]: paramsBaseSchema,
  [Segments.BODY]: {
    name: Joi.string().optional(),
  },
})

const validateRemoveExampleSchema = celebrate({
  [Segments.PARAMS]: paramsBaseSchema,
})

const validateExampleExistence = async (
  request: Request,
  _: Response,
  next: NextFunction,
) => {
  const { db, messages, params, query, body } = request

  const exampleID =
    params.exampleID || query.exampleID || body.exampleID || body.example?._id

  if (isNullOrUndefined(exampleID)) {
    return next()
  }

  const example = await findExampleByID(db, exampleID)

  if (isNullOrUndefined(example)) {
    throw new AppError(messages.errors.examples[404], null, HttpStatus.NotFound)
  }

  request.locals.example = example

  return next()
}

const validateExamplesExistence = async (
  request: Request,
  _: Response,
  next: NextFunction,
) => {
  const { db, messages, query, body } = request

  const examplesIDs = query.exampleIDs || body.exampleIDs || body.examples?.map(p => p._id)

  if (isNullOrUndefined(examplesIDs)) {
    return next()
  }

  // isEmpty
  if (isEmpty(examplesIDs)) {
    request.locals.examples = []
    return next()
  }

  const examples = await findExamplesByIDs(db, examplesIDs)

  // Contains Not found examples
  if (isNullOrUndefined(examples) || examples.length !== examplesIDs.length) {
    throw new AppError(messages.errors.examples[404], null, HttpStatus.NotFound)
  }

  request.locals.examples = examples

  return next()
}

export {
  validateCreateExampleSchema,
  validateFetchExamplesSchema,
  validateFetchExampleSchema,
  validateEditExampleSchema,
  validateRemoveExampleSchema,
  validateExampleExistence,
  validateExamplesExistence,
}
