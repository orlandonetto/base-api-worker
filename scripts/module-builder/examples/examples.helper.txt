import { ObjectId } from 'mongodb'

import { TypeExample } from './examples.types'

const mapExample = (body: TypeExample) => {
  return {
    deleted: false,
    ...body,
    createdAt: new Date(),
    updatedAt: new Date(),
  }
}

const mapExampleRef = (example: TypeExample) => {
  return {
    _id: new ObjectId(example._id.toString()),
    name: example.name,
  }
}

export { mapExample, mapExampleRef }
